cmake_minimum_required(VERSION 3.21)
project(PhysicsHostValidation LANGUAGES C CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(default_components "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(PHYSICS_RT_COMPONENT_ROOT "${default_components}" CACHE PATH "Directory containing VxMath, CK2, RenderEngine and BuildingBlocks")
if (WIN32)
    add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
endif ()
foreach (component VXMATH CK2 CKRE)
    set(${component}_BUILD_STATIC ON CACHE BOOL "")
    set(${component}_BUILD_TESTS OFF CACHE BOOL "")
    set(${component}_INSTALL OFF CACHE BOOL "")
endforeach ()
set(VXMATH_BUILD_SHARED ON CACHE BOOL "")
set(CK2_BUILD_SHARED ON CACHE BOOL "")
set(CKRE_BUILD_SHARED OFF CACHE BOOL "")
set(CKRE_BUILD_SDL_GPU_RASTERIZER OFF CACHE BOOL "")
set(CKRE_BUILD_TOOLS OFF CACHE BOOL "")
set(CKBB_BUILD_ALL_MODULES OFF CACHE BOOL "")
set(CKBB_BUILD_physics_RT ON CACHE BOOL "")
set(CKBB_BUILD_SHARED ON CACHE BOOL "")
set(CKBB_BUILD_STATIC ON CACHE BOOL "")
set(CKBB_BUILD_TESTS ON CACHE BOOL "")
set(CKBB_INSTALL OFF CACHE BOOL "")
foreach (component VxMath CK2 RenderEngine BuildingBlocks)
    if (NOT EXISTS "${PHYSICS_RT_COMPONENT_ROOT}/${component}/CMakeLists.txt")
        message(FATAL_ERROR "Set PHYSICS_RT_COMPONENT_ROOT to the directory containing VxMath, CK2, RenderEngine and BuildingBlocks")
    endif ()
    add_subdirectory("${PHYSICS_RT_COMPONENT_ROOT}/${component}" "${component}")
endforeach ()
